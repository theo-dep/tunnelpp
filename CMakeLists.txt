cmake_minimum_required(VERSION 3.26)
project(tunnelpp VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(TUNNELPP_BUILD_SERVER "Build the tunnel server" ON)
option(TUNNELPP_BUILD_CLIENT "Build the tunnel client" ON)
option(TUNNELPP_BUILD_TESTS "Build the test suite" OFF)
option(TUNNELPP_ENABLE_SANITIZERS "Build with AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(TUNNELPP_BUILD_COVERAGE "Enable clang source-based coverage instrumentation" OFF)

find_package(httplib CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(spdlog CONFIG REQUIRED)

if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-Wall -Wextra -Werror -pedantic)
    # sanitizers do not link on Windows with clang (windows-msvc target: no UBSan runtime with -fsanitize=address,undefined)
    if(TUNNELPP_ENABLE_SANITIZERS AND NOT WIN32)
        add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
        add_link_options(-fsanitize=address,undefined)
    endif()
    if(TUNNELPP_BUILD_COVERAGE)
        if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
            add_compile_options(-fprofile-instr-generate -fcoverage-mapping -O0 -g)
            add_link_options(-fprofile-instr-generate -fcoverage-mapping)
        else()
            add_compile_options(--coverage -O0 -g)
            add_link_options(--coverage)
        endif()
    endif()
elseif(MSVC)
    add_compile_options(/Wall /WX)
endif()

if(TUNNELPP_BUILD_TESTS)
    enable_testing()
    find_package(Catch2 CONFIG REQUIRED)
endif()

add_subdirectory(common)
if(TUNNELPP_BUILD_SERVER)
    add_subdirectory(server)
endif()
if(TUNNELPP_BUILD_CLIENT)
    add_subdirectory(client)
endif()
if(TUNNELPP_BUILD_TESTS AND TUNNELPP_BUILD_SERVER AND TUNNELPP_BUILD_CLIENT)
    add_subdirectory(tests)
endif()
